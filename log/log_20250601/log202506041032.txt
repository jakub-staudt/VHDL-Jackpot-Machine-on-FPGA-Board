entity CLK_DIV_F is
    Port ( CLKin : in STD_LOGIC;
           CLKout : out STD_LOGIC);
end CLK_DIV_F;

architecture Behavioral of CLK_DIV_f is

signal clkBUFF: std_logic_vector(13 downto 0) := (others => '0');

begin
process(CLKin)
begin
if(rising_edge(CLKin)) then
clkBUFF <= clkBUFF + 1;
end if;
end process;
CLKout<=clkBUFF(13);
end Behavioral;
