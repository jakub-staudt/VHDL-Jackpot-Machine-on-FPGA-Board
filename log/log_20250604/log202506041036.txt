entity fullbitadder is
    Port( A: in std_logic;
    B: in std_logic;
    Cin: in std_logic;
    S: out std_logic;
    Cout: out std_logic);
end fullbitadder;

architecture Behavioral of fullbitadder is

begin

S<=(A xor B) xor Cin;
Cout<=(A and B) or (Cin and (A xor B));

end Behavioral;
